tests = [
    ("Attention Statistics", test_attention_statistics),
    ("Compression Results", test_compression_results),
    ("Benchmark Trends", test_benchmark_trends)
]

all_passed = True
for test_name, test_func in tests:
    try:
        if test_func():

        all_passed = False

    print("✅ ALL TESTS PASSED - Reproduction successful!")
else:
